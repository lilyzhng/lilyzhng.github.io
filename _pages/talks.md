<script>
document.addEventListener('DOMContentLoaded', function () {
  var buttons = document.querySelectorAll('.posts-filter .filter-item');
  var cards = document.querySelectorAll('.talk-card');
  buttons.forEach(function (btn) {
    btn.addEventListener('click', function () {
      buttons.forEach(function (b) { b.classList.remove('is-active'); });
      btn.classList.add('is-active');
      var f = btn.getAttribute('data-filter');
      cards.forEach(function (c) {
        c.style.display = (f === 'all' || c.getAttribute('data-type') === f) ? '' : 'none';
      });
    });
  });

  // photo carousels: swipe (native scroll-snap) + arrows + dots
  document.querySelectorAll('.talk-carousel-wrap').forEach(function (wrap) {
    var track = wrap.querySelector('.talk-carousel');
    var imgs = track.querySelectorAll('img');
    var dotsBox = wrap.querySelector('.car-dots');
    var dots = [];
    imgs.forEach(function (_, i) {
      var d = document.createElement('span');
      d.className = 'car-dot' + (i === 0 ? ' is-on' : '');
      dotsBox.appendChild(d);
      dots.push(d);
    });
    function current() { return Math.round(track.scrollLeft / track.clientWidth); }
    function goTo(i) {
      i = Math.max(0, Math.min(imgs.length - 1, i));
      track.scrollTo({ left: i * track.clientWidth, behavior: 'smooth' });
    }
    track.addEventListener('scroll', function () {
      var i = current();
      dots.forEach(function (d, j) { d.classList.toggle('is-on', j === i); });
    }, { passive: true });
    wrap.querySelector('.car-prev').addEventListener('click', function (e) {
      e.preventDefault(); goTo(current() - 1);
    });
    wrap.querySelector('.car-next').addEventListener('click', function (e) {
      e.preventDefault(); goTo(current() + 1);
    });
    track.addEventListener('keydown', function (e) {
      if (e.key === 'ArrowLeft') { e.preventDefault(); goTo(current() - 1); }
      if (e.key === 'ArrowRight') { e.preventDefault(); goTo(current() + 1); }
    });
  });

  // touch devices: reveal color as cards scroll into view (same as Projects)
  if (window.matchMedia('(hover: none)').matches && 'IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) { e.target.classList.toggle('in-view', e.intersectionRatio >= 0.6); });
    }, { threshold: [0, 0.6, 1] });
    cards.forEach(function (c) { io.observe(c); });
  }
});
</script>
